function safe(fn) { try { fn(); } catch (e) { Logger.log(`Data fetch warning: ${e.message}`); } }

/** FRED — returns latest observation. Free API key required. */
function fredLatest(seriesId, label, unit) {
  const url = `https://api.stlouisfed.org/fred/series/observations`
    + `?series_id=${seriesId}&api_key=${CONFIG.FRED_API_KEY}`
    + `&file_type=json&sort_order=desc&limit=2`;
  const resp = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
  if (resp.getResponseCode() !== 200) throw new Error(`FRED ${seriesId} HTTP ${resp.getResponseCode()}`);
  const j = JSON.parse(resp.getContentText());
  const obs = (j.observations || []).filter(o => o.value !== ".");
  if (obs.length === 0) throw new Error(`FRED ${seriesId} no data`);
  const latest = parseFloat(obs[0].value);
  const prev   = obs[1] ? parseFloat(obs[1].value) : latest;
  return { label, value: latest, prev: prev, change: latest - prev, unit, date: obs[0].date };
}

/** Yahoo Finance public quote endpoint — no key needed. */
function yahooQuote(symbol, label, unit) {
  const url = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(symbol)}?interval=1d&range=5d`;
  const resp = UrlFetchApp.fetch(url, {
    muteHttpExceptions: true,
    headers: { "User-Agent": "Mozilla/5.0 GlobalBriefingBot/3.0" }
  });
  if (resp.getResponseCode() !== 200) throw new Error(`Yahoo ${symbol} HTTP ${resp.getResponseCode()}`);
  const j = JSON.parse(resp.getContentText());
  const result = j.chart && j.chart.result && j.chart.result[0];
  if (!result) throw new Error(`Yahoo ${symbol} no result`);
  const closes = result.indicators.quote[0].close.filter(v => v !== null);
  if (closes.length < 1) throw new Error(`Yahoo ${symbol} no closes`);
  const latest = closes[closes.length - 1];
  const prev   = closes.length >= 2 ? closes[closes.length - 2] : latest;
  return {
    label, value: latest, prev, change: latest - prev,
    pctChange: prev ? ((latest - prev) / prev) * 100 : 0,
    unit, date: new Date(result.meta.regularMarketTime * 1000).toISOString().slice(0, 10),
    symbol
  };
}

function formatMarketData(d) {
  const lines = ["[LIVE MARKET DATA — all values as of latest market close/intraday]"];
  const fmt = (v) => {
    if (!v) return null;
    const val = typeof v.value === "number" ? v.value.toFixed(2) : v.value;
    const chg = v.pctChange !== undefined
      ? `${v.pctChange >= 0 ? "+" : ""}${v.pctChange.toFixed(2)}%`
      : `${v.change >= 0 ? "+" : ""}${v.change.toFixed(2)}`;
    const datePart = v.date ? ` [${v.date}]` : "";
    return `  ${v.label}: ${val}${v.unit} (${chg})${datePart}`;
  };
  Object.values(d).forEach(v => {
    const line = fmt(v);
    if (line) lines.push(line);
  });
  return lines.join("\n");
}

// ------------------------------------------------------------

function fetchPortfolioData() {
  const data = {};
  for (const pos of CONFIG.PORTFOLIO) {
    data[pos.label] = { label: pos.label, thesis: pos.thesis, quotes: [] };
    for (const sym of pos.tickers) {
      try {
        const q = yahooQuote(sym, sym, "");
        data[pos.label].quotes.push(q);
      } catch (e) {
        Logger.log(`Portfolio fetch warning [${sym}]: ${e.message}`);
        data[pos.label].quotes.push({ symbol: sym, label: sym, error: e.message });
      }
    }
  }
  return data;
}

function formatPortfolioData(p) {
  const lines = ["[YOUR PORTFOLIO — LIVE QUOTES + THESIS]"];
  for (const key in p) {
    const pos = p[key];
    lines.push(`\n• ${pos.label}`);
    lines.push(`  Thesis: ${pos.thesis}`);
    const quoteStrs = pos.quotes.map(q => {
      if (q.error) return `${q.symbol}=ERR`;
      const val = typeof q.value === "number" ? q.value.toFixed(2) : q.value;
      const chg = q.pctChange !== undefined ? `${q.pctChange >= 0 ? "+" : ""}${q.pctChange.toFixed(2)}%` : "";
      const datePart = q.date ? ` as of ${q.date}` : "";
      return `${q.symbol}=${val} (${chg})${datePart}`;
    });
    lines.push(`  Live: ${quoteStrs.join(" | ")}`);
  }
  return lines.join("\n");
}
